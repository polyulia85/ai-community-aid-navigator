import json
import re
from typing import Any, Dict, Optional, Tuple

_RE_PII = re.compile(
    r"\b\d{3}[-\s]\d{2}[-\s]\d{4}\b|\b\d{9}\b|\b(?:\d{4}[-\s]?){3}\d{4}\b"
    r"|\b(?:routing|account)[\s#:]+\d{8,12}\b",
    re.IGNORECASE,
)
_PROHIBITED_FIELD_NAMES = [
    "ssn", "social_security", "bank_account", "routing_number", "credit_card", "cvv"
]
_REQUIRED_FIELDS = {"request_id", "run_id", "case_id", "case_version", "correlation_id", "agent_id", "schema_version", "status"}


class SupervisorValidator:
    PROHIBITED_PII = _PROHIBITED_FIELD_NAMES

    @classmethod
    def validate(cls, req: Dict[str, Any], res: Dict[str, Any]) -> Tuple[bool, Optional[str]]:
        # Schema check: required fields present
        missing = _REQUIRED_FIELDS - res.keys()
        if missing:
            return False, f"SCHEMA_INVALID: Missing fields: {', '.join(sorted(missing))}"

        # Identity checks
        if res.get("request_id") != req.get("request_id"):
            return False, "IDENTITY_MISMATCH: request_id mismatch"
        if res.get("case_id") != req.get("case_id"):
            return False, "IDENTITY_MISMATCH: case_id mismatch"
        if res.get("correlation_id") != req.get("correlation_id"):
            return False, "IDENTITY_MISMATCH: correlation_id mismatch"
        if res.get("case_version") != req.get("case_version"):
            return False, "STALE_RESULT: Case version mismatch"

        # Privacy check: prohibited field names and PII patterns in serialized output
        serialized = json.dumps(res).lower()
        for pii in cls.PROHIBITED_PII:
            if pii in serialized:
                return False, f"PRIVACY_VIOLATION: Prohibited PII field '{pii}' detected"
        if _RE_PII.search(json.dumps(res)):
            return False, "PRIVACY_VIOLATION: Raw PII pattern detected in response"

        return True, None
